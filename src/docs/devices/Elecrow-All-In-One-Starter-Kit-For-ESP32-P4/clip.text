### Advanced: 7-Inch Display Configuration

The 7-inch IPS display (1024×600) uses MIPI-DSI interface with **EK79007 controller** (verified from official Elecrow code):

```yaml
# External component for MIPI-DSI support
external_components:
  - source: github://pr#11886
    components: [mipi_dsi]
    refresh: 1h

# Font for LVGL
font:
  - file: gfonts://Roboto
    id: main_font
    size: 24
    glyphs: "!\"#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\\]^_`abcdefghijklmnopqrstuvwxyz{|}~ "

# Color definitions
color:
  - id: COLOR_BLACK
    red: 0%
    green: 0%
    blue: 0%
  - id: COLOR_WHITE
    red: 100%
    green: 100%
    blue: 100%

# MIPI-DSI Display Configuration (EK79007 Controller)
# Parameters verified from official Elecrow ESP-IDF code
display:
  - platform: mipi_dsi
    id: main_display
    model: CUSTOM  # EK79007 controller
    rotation: 0  # Set to 180 if needed
    update_interval: never
    auto_clear_enabled: false
    lanes: 2
    lane_bit_rate: 900Mbps  # Official: 900 Mbps
    color_order: rgb
    pclk_frequency: 51MHz   # Official: 51 MHz DPI clock
    dimensions:
      width: 1024
      height: 600
    # Timing parameters from official code
    hsync_back_porch: 160
    hsync_pulse_width: 70
    hsync_front_porch: 160
    vsync_back_porch: 23
    vsync_pulse_width: 10   # Official code uses 10, not 1
    vsync_front_porch: 12
    # Display initialization sequence
    # Note: EK79007 init is handled internally by the driver
    init_sequence:
      - [0xB2, 0x10]
      - [0x80, 0x8B]
      - [0x81, 0x78]
      - [0x82, 0x84]
      - [0x83, 0x88]
      - [0x84, 0xA8]
      - [0x85, 0xE3]
      - [0x86, 0x88]
      - delay: 120ms
      - [0x29]  # Display on

# GPIO Control for Display
# UPDN: GPIO32 (vertical scan direction: 0=normal, 1=inverted)
# SHLR: GPIO33 (horizontal scan direction: 0=normal, 1=inverted)
switch:
  - platform: gpio
    pin: GPIO32
    name: "Display UPDN"
    id: display_updn
    internal: true
    restore_mode: RESTORE_DEFAULT_OFF
  
  - platform: gpio
    pin: GPIO33
    name: "Display SHLR"
    id: display_shlr
    internal: true
    restore_mode: RESTORE_DEFAULT_ON  # Official code sets this to 1

# LVGL Configuration for UI
lvgl:
  buffer_size: 100%
  byte_order: little_endian
  displays:
    - main_display
  touchscreens:
    - touch
  pages:
    - id: main_page
      widgets:
        - label:
            align: TOP_MID
            y: 20
            text: "ESP32-P4 Control Panel"
            text_font: main_font
            text_color: 0xFFFFFF
        
        - button:
            align: CENTER
            width: 200
            height: 60
            widgets:
              - label:
                  align: CENTER
                  text: "Test Button"
                  text_font: main_font
```

  framework:
    sdkconfig_options:
      # Enable SC2336 camera driver
      CONFIG_CAMERA_SC2336: "y"

      CONFIG_CAMERA_SC2336_AUTO_DETECT: "y"
      CONFIG_CAMERA_SC2336_AUTO_DETECT_MIPI_INTERFACE_SENSOR: "y"

      name: "Camera Hue"

      name: "Camera Saturation"
    filter:
      name: "Camera Filter"
    red:
      name: "Camera Red"
    green:
      name: "Camera Green"
    blue:
      name: "Camera Blue"

  name: "ESP32-P4 Camera"
